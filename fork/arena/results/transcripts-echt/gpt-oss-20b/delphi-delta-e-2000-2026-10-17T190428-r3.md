  X := 0.4124564 * rLin + 0.3575761 * gLin + 0.1804375 * bLin;
  Y := 0.2126729 * rLin + 0.7151522 * gLin + 0.0721750 * bLin;
  Z := 0.0193339 * rLin + 0.1191920 * gLin + 0.9503041 * bLin;

  // XYZ → Lab
  if X / D65_Xn > 0.008856 then
    fx := Power(X / D65_Xn, 1 / 3)
  else
    fx := 7.787 * (X / D65_Xn) + 16 / 116;

  if Y / D65_Yn > 0.008856 then
    fy := Power(Y / D65_Yn, 1 / 3)
  else
    fy := 7.787 * (Y / D65_Yn) + 16 / 116;

  if Z / D65_Zn > 0.008856 then
    fz := Power(Z / D65_Zn, 1 / 3)
  else
    fz := 7.787 * (Z / D65_Zn) + 16 / 116;

  L   := 116 * fy - 16;
  aLab:= 500 * (fx - fy);
  bLab:= 200 * (fy - fz);

  Result.L    := L;
  Result.a    := aLab;
  Result.bLab := bLab;
end;

{--- 4. Helper: degrees to radians ----------------------------}
function DegToRad(d: Double): Double;
begin
  Result := d * PI / 180.0;
end;

{--- 5. Helper: hue angle [0,360) --------------------------------}
function HueAngle(a, b: Double): Double;
var
  hRad: Double;
begin
  hRad := ArcTan2(b, a);
  Result := hRad * 180 / PI;
  if Result < 0 then
    Result := Result + 360;
end;

{--- 6. CIEDE2000 ------------------------------------------------}
function CIEDE2000(Lab1, Lab2: TLab): Double;
var
  // Means
  LPrimeMean, CPrimeMean, hPrimeMean: Double;
  // Shifted a
  a1p, a2p: Double;
  // Chromaticities
  C1p, C2p: Double;
  // Delta components
  dLStar, dCp, dHp: Double;
  // Weights
  SL, SC, SH: Double;
  // Rotation term
  RT: Double;
  // Intermediate
  h1p, h2p, hDiff: Double;
  T, deltaTheta: Double;
  GShift, sq1, sq2: Double;
begin
  // ---- 1. Mean of L' ------------------------------------
  LPrimeMean := (Lab1.L + Lab2.L) / 2.0;

  // ---- 2. Compute a' ---------------------------------
  GShift := 0.5 * (1 - Sqrt(Power((CPrimeMean)/2.0, 7) /
                           (Power((CPrimeMean)/2.0, 7) + 6103515625)));

  a1p := (1 + GShift) * Lab1.a;
  a2p := (1 + GShift) * Lab2.a;

  // ---- 3. Compute C' -----------------------------------
  C1p := Sqrt(a1p*a1p + Lab1.bLab*Lab1.bLab);
  C2p := Sqrt(a2p*a2p + Lab2.bLab*Lab2.bLab);
  CPrimeMean := (C1p + C2p) / 2.0;

  // ---- 4. Compute h' angles ----------------------------
  h1p := HueAngle(a1p, Lab1.bLab);
  h2p := HueAngle(a2p, Lab2.bLab);

  // ---- 5. Delta components -----------------------------
  dLStar := Lab2.L - Lab1.L;
  dCp    := C2p - C1p;

  if (C1p * C2p = 0) then
    dHp := 0
  else
  begin
    hDiff := h2p - h1p;
    if Abs(hDiff) > 180 then
      if hDiff > 0 then
        hDiff := hDiff - 360
      else
        hDiff := hDiff + 360;
    dHp := 2 * Sqrt(C1p * C2p) * Sin(DegToRad(hDiff / 2));
  end;

  // ---- 6. Means for weighting -------------------------
  if (C1p * C2p = 0) then
    hPrimeMean := h1p + h2p
  else
  begin
    if Abs(h1p - h2p) <= 180 then
      hPrimeMean := (h1p + h2p) / 2
    else
    begin
      if h1p + h2p < 360 then
        hPrimeMean := (h1p + h2p + 360) / 2
      else
        hPrimeMean := (h1p + h2p - 360) / 2;
    end;
  end;

  // ---- 7. Weights --------------------------------------
  SL := 1 + 0.015 * Power(LPrimeMean - 50, 2) /
        Sqrt(20 + Power(LPrimeMean - 50, 2));

  SC := 1 + 0.045 * CPrimeMean;

  T := 1 - 0.17 * Cos(DegToRad(hPrimeMean - 30))
      + 0.24 * Cos(DegToRad(2 * hPrimeMean))
      + 0.32 * Cos(DegToRad(3 * hPrimeMean + 6))
      - 0.20 * Cos(DegToRad(4 * hPrimeMean - 63));

  SH := 1 + 0.015 * CPrimeMean * T;

  // ---- 8. Rotation term ----------------------------------
  deltaTheta := 30 * Exp(-Power((hPrimeMean - 275) / 25.0, 2));
  RT := -2 * Sin(DegToRad(2 * deltaTheta)) * T;

  // ---- 9. Final DeltaE ------------------------------------
  Result := Sqrt(
    Power(dLStar / SL, 2) +
    Power(dCp    / SC, 2) +
    Power(dHp    / SH, 2)
  );
end;

{--- 7. Public function ---------------------------------------}
function DeltaE2000(R1, G1, B1, R2, G2, B2: Byte): Double;
var
  Lab1, Lab2: TLab;
begin
  Lab1 := RGBtoLab(R1, G1, B1);
  Lab2 := RGBtoLab(R2, G2, B2);
  Result := CIEDE2000(Lab1, Lab2);
end;

end.
