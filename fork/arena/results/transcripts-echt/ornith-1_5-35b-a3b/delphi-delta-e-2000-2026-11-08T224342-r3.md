uses
  SysUtils, Math;

function DeltaE2000(R1, G1, B1, R2, G2, B2: Byte): Double;
var
  Px, Py, Pz: Double;
  LL1, AA1, BB1, LL2, AA2, BB2: Double;
  C1p, C2p, CP1, CP2, HP1, HP2, G: Double;
  dLp, dCp, dH, dHp, hbar, fp, theta, RC, SC, SH, RT: Double;

  // sRGB-Kanal (0..1) in linearen Licht umrechnen
  function sRGB2Lin(c: Double): Double;
  begin
    if c <= 0.04045 then
      Result := c / 12.92
    else
      Result := Power((c + 0.055) / 1.055, 2.4);
  end;

  // sRGB (0..255) nach XYZ (Weißpunkt D65)
  function RGB2XYZ(r, g, b: Double; var x, y, z: Double): Boolean;
  begin
    r := sRGB2Lin(r / 255);
    g := sRGB2Lin(g / 255);
    b := sRGB2Lin(b / 255);
    x := r * 0.4124 + g * 0.3576 + b * 0.1805;
    y := r * 0.2126 + g * 0.7152 + b * 0.0722;
    z := r * 0.0193 + g * 0.1192 + b * 0.9505;
    Result := True;
  end;

  // XYZ (D65) nach CIELAB
  function XYZ2Lab(x, y, z: Double; var l, a, b: Double): Boolean;
  const
    xn = 0.95047;
    yn = 1.00000;
    zn = 1.08883;
    t0 = 0.008856;
  function f(t: Double): Double;
  begin
    if t > t0 then
      Result := Power(t, 1 / 3)
    else
      Result := 7.787 * t + 16 / 116;
  end;
  begin
    l := 116 * f(y / yn) - 16;
    a := 500 * (f(x / xn) - f(y / yn));
    b := 200 * (f(y / yn) - f(z / zn));
    Result := True;
  end;

begin
  Result := 0;

  if not RGB2XYZ(R1, G1, B1, Px, Py, Pz) then
    Exit;
  if not XYZ2Lab(Px, Py, Pz, LL1, AA1, BB1) then
    Exit;
  if not RGB2XYZ(R2, G2, B2, Px, Py, Pz) then
    Exit;
  if not XYZ2Lab(Px, Py, Pz, LL2, AA2, BB2) then
    Exit;

  // ----- CIEDE2000 (kL = kC = kH = 1) -----
  C1p := Sqrt(AA1 * AA1 + BB1 * BB1);
  C2p := Sqrt(AA2 * AA2 + BB2 * BB2);

  if (Power(C1p, 7) + Power(C2p, 7) = 0) then
    G := 0
  else
    G := 0.5 * (1 - Sqrt(Power(C1p, 7) / (Power(C1p, 7) + Power(C2p, 7))));

  AA1 := (1 + G) * AA1;
  AA2 := (1 + G) * AA2;

  CP1 := Sqrt(AA1 * AA1 + BB1 * BB1);
  CP2 := Sqrt(AA2 * AA2 + BB2 * BB2);

  HP2 := Arctan2(BB2, AA2) * 180 / Pi;
  if HP2 < 0 then
    HP2 := HP2 + 360;

  dLp := LL2 - LL1;
  dCp := CP2 - CP1;

  if (CP1 * CP2 = 0) then
    dH := 0
  else if Abs(HP2 - HP1) <= 180 then
    dH := HP2 - HP1
  else if (HP2 - HP1) > 180 then
    dH := HP2 - HP1 - 360
  else
    dH := HP2 - HP1 + 360;

  dHp := 2 * Sqrt(CP1 * CP2) * Sin(dH / 2 * Pi / 180);

  if (CP1 * CP2 = 0) then
    hbar := HP1 + HP2
  else if Abs(HP1 - HP2) <= 180 then
    hbar := (HP1 + HP2) / 2
  else if (HP1 + HP2) < 360 then
    hbar := (HP1 + HP2 + 360) / 2
  else
    hbar := (HP1 + HP2 - 360) / 2;

  fp := 1 - 0.071 * Cos((hbar - 15) * Pi / 180);
  theta := 30 * Exp(-Power((hbar - 275) / 25, 2));

  if (Power(CP1, 7) + Power(CP2, 7) = 0) then
    RC := 0
  else
    RC := 2 * Sqrt(Power(CP1, 7) / (Power(CP1, 7) + Power(CP2, 7))) * Sin(theta * Pi / 180);

  SH := 1 + 0.015 * fp * Cos((hbar - theta) * Pi / 180);
  RT := -Sin(2 * theta * Pi / 180) * RC;

  dCp := dCp / SC;
  dHp := dHp / SH;

  Result := Sqrt(Power(dLp, 2) + Power(dCp, 2) + Power(dHp, 2) + RT * dCp * dHp);
