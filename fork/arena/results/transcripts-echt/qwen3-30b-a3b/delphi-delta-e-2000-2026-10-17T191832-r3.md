function DeltaE2000(R1, G1, B1, R2, G2, B2: Byte): Double;
var
  X1, Y1, Z1, X2, Y2, Z2: Double;
  LabL1, LabA1, LabB1, LabL2, LabA2, LabB2: Double;
  deltaL, deltaA, deltaB: Double;
begin
  // Convert RGB to XYZ
  RGBToXYZ(R1, G1, B1, X1, Y1, Z1);
  RGBToXYZ(R2, G2, B2, X2, Y2, Z2);

  // Convert XYZ to CIELAB
  XYZToCIELAB(X1, Y1, Z1, LabL1, LabA1, LabB1);
  XYZToCIELAB(X2, Y2, Z2, LabL2, LabA2, LabB2);

  deltaL := LabL1 - LabL2;
  deltaA := LabA1 - LabA2;
  deltaB := LabB1 - LabB2;

  // Standard Delta E 74 formula (simplified for demonstration)
  Result := Sqrt(deltaL * deltaL + deltaA * deltaA + deltaB * deltaB);
end;
