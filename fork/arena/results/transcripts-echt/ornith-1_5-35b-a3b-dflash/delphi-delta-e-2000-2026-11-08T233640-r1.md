  D65_Xn = 0.95047;

  D65_Zn = 1.08883;

  EPS    = 0.008856451617;   // = 216/24389

function LabF(t: Double): Double;
// f(t) aus der CIE-Definition für die Umrechnung XYZ -> CIELAB
begin
  if t > EPS then
    Result := Power(t, 1.0 / 3.0)
  else
    Result := t / KAPPA + 4.0 / 29.0;
end;

function sRGBToLinear(c: Double): Double;
// c im Bereich [0,1]; kehrt die sRGB-Gamma-Korrektur um
begin
  if c > 0.04045 then
    Result := Power((c + 0.055) / 1.055, 2.4)
  else
    Result := c / 12.92;
end;

function XYZToLab(X, Y, Z: Double; var L, a, b: Double): Boolean;
var
  fx, fy, fz: Double;
begin

  fy := LabF(Y / D65_Yn);
  fz := LabF(Z / D65_Zn);
  L := 116.0 * fy - 16.0;
  a := 500.0 * (fx - fy);
  b := 200.0 * (fy - fz);

end;
