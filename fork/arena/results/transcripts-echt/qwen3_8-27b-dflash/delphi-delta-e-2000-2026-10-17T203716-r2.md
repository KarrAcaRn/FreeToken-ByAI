uses
  SysUtils, Math;

function SRGBToLinear(const Val: Byte): Double;
var
  f: Double;
begin
  f := Val / 255.0;
  if f <= 0.04045 then
    Result := f / 12.92
  else
    Result := Power((f + 0.055) / 1.055, 2.4);
end;

procedure LinearRGBtoXYZ(const R, G, B: Double; out X, Y, Z: Double);
begin
  X :=  0.4124564 * R + 0.3575761 * G + 0.1804375 * B;
  Y :=  0.2126729 * R + 0.7373822 * G + 0.0701449 * B;
  Z :=  0.0193339 * R + 0.1191920 * G + 0.9503041 * B;
end;

procedure XYZtoCIELab(const X, Y, Z: Double; out L, A, B: Double);
const
  Xn  = 0.95047;
  Yn  = 1.00000;
  Zn  = 1.08883;
  Eps = 216.0 / 24389.0;
  K   = 24389.0 / 27.0;
var
  xr, yr, zr: Double;
  fx, fy, fz: Double;
begin
  xr := X / Xn;
  yr := Y / Yn;
  zr := Z / Zn;

  if xr > Eps then
    fx := Power(xr, 1.0 / 3.0)
  else
    fx := (K * xr + 4.0) / 29.0;

  if yr > Eps then
    fy := Power(yr, 1.0 / 3.0)
  else
    fy := (K * yr + 4.0) / 29.0;

  if zr > Eps then
    fz := Power(zr, 1.0 / 3.0)
  else
    fz := (K * zr + 4.0) / 29.0;

  L := 116.0 * fy - 16.0;
  A := 500.0 * (fx - fy);
  B := 200.0 * (fy - fz);
end;

function CIEDE2000(const L1, a1, b1, L2, a2, b2: Double): Double;
const
  Deg2Rad = Pi / 180.0;
  Rad2Deg = 180.0 / Pi;
var
  C1, C2, Cbar, G: Double;
  a1p, a2p, C1p, C2p: Double;
  h1p, h2p, hbar: Double;
  dLp, dCp, dh, dHp: Double;
  Lbar, Cbarp: Double;
  T, dTheta, RC: Double;
  SL, SC, SH, RT: Double;
  t1, t2, t3: Double;
begin
  C1   := Sqrt(Sqr(a1) + Sqr(b1));
  C2   := Sqrt(Sqr(a2) + Sqr(b2));
  Cbar := (C1 + C2) * 0.5;

  G := 0.5 * (1.0 - Sqrt(Power(Cbar, 7.0) /
       (Power(Cbar, 7.0) + Power(25.0, 7.0))));

  a1p := a1 * (1.0 + G);
  a2p := a2 * (1.0 + G);

  C1p := Sqrt(Sqr(a1p) + Sqr(b1));
  C2p := Sqrt(Sqr(a2p) + Sqr(b2));

  h1p := ArcTan2(b1, a1p) * Rad2Deg;
  if h1p < 0.0 then h1p := h1p + 360.0;
  h2p := ArcTan2(b2, a2p) * Rad2Deg;
  if h2p < 0.0 then h2p := h2p + 360.0;

  dLp := L2 - L1;
  dCp := C2p - C1p;

  if C1p * C2p = 0.0 then
    dh := 0.0
  else begin
    dh := h2p - h1p;
    if Abs(dh) <= 180.0 then
      dh := dh
    else if dh > 180.0 then
      dh := dh - 360.0
    else
      dh := dh + 360.0;
  end;

  dHp := 2.0 * Sqrt(C1p * C2p) * Sin(dh * 0.5 * Deg2Rad);

  Lbar  := (L1 + L2) * 0.5;
  Cbarp := (C1p + C2p) * 0.5;

  if C1p * C2p = 0.0 then
    hbar := h1p + h2p
  else begin
    if Abs(h1p - h2p) <= 180.0 then
      hbar := (h1p + h2p) * 0.5
    else if (h1p + h2p) < 360.0 then
      hbar := (h1p + h2p) * 0.5 + 180.0
    else
      hbar := (h1p + h2p) * 0.5 - 180.0;
  end;

  T := 1.0
    - 0.17 * Cos((hbar -  30.0) * Deg2Rad)
    + 0.24 * Cos( 2.0 * hbar    * Deg2Rad)
    + 0.32 * Cos((3.0 * hbar +  6.0) * Deg2Rad)
    - 0.20 * Cos((4.0 * hbar - 63.0) * Deg2Rad);

  dTheta := 30.0 * Exp(-Sqr((hbar - 275.0) / 25.0));

  RC := 2.0 * Sqrt(Power(Cbarp, 7.0) /
        (Power(Cbarp, 7.0) + Power(25.0, 7.0)));

  SL := 1.0 + 0.015 * Sqr(Lbar - 50.0) /
        Sqrt(20.0 + Sqr(Lbar - 50.0));
  SC := 1.0 + 0.045 * Cbarp;
  SH := 1.0 + 0.015 * Cbarp * T;

  RT := -Sin(2.0 * dTheta * Deg2Rad) * RC;

  t1 := dLp / SL;
  t2 := dCp / SC;
  t3 := dHp / SH;

  Result := Sqrt(Sqr(t1) + Sqr(t2) + Sqr(t3) + RT * t2 * t3);
end;

function DeltaE2000(R1, G1, B1, R2, G2, B2: Byte): Double;
var
  lr1, lg1, lb1, lr2, lg2, lb2: Double;
  x1, y1, z1, x2, y2, z2: Double;
  L1, A1, B1lab, L2, A2, B2lab: Double;
begin
  lr1 := SRGBToLinear(R1);
  lg1 := SRGBToLinear(G1);
  lb1 := SRGBToLinear(B1);
  lr2 := SRGBToLinear(R2);
  lg2 := SRGBToLinear(G2);
  lb2 := SRGBToLinear(B2);

  LinearRGBtoXYZ(lr1, lg1, lb1, x1, y1, z1);
  LinearRGBtoXYZ(lr2, lg2, lb2, x2, y2, z2);

  XYZtoCIELab(x1, y1, z1, L1, A1, B1lab);
  XYZtoCIELab(x2, y2, z2, L2, A2, B2lab);

  Result := CIEDE2000(L1, A1, B1lab, L2, A2, B2lab);
end;
