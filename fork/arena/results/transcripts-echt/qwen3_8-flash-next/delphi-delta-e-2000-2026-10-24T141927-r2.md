uses
  Math;

function DeltaE2000(R1, G1, B1, R2, G2, B2: Byte): Double;

implementation

type
  TLab = record
    L, a, b: Double;
  end;

function SRGBToLinear(C: Double): Double;
begin
  C := C / 255.0;
  if C <= 0.04045 then
    Result := C / 12.92
  else
    Result := Power((C + 0.055) / 1.055, 2.4);
end;

function LinearRGBToLab(Rlin, Glin, Blin: Double): TLab;
const
  Xn = 95.047;
  Yn = 100.0;
  Zn = 108.883;
  Eps = 216.0 / 24389.0;
  Kap = 24389.0 / 27.0;
var
  X, Y, Z, fx, fy, fz: Double;

  function F(T: Double): Double;
  begin
    if T > Eps then
      Result := Power(T, 1.0 / 3.0)
    else
      Result := (Kap * T + 16.0) / 116.0;
  end;

begin
  X := (0.4124564 * Rlin + 0.3575761 * Glin + 0.1804375 * Blin) * 100.0;
  Y := (0.2126729 * Rlin + 0.7151522 * Glin + 0.0721750 * Blin) * 100.0;
  Z := (0.0193339 * Rlin + 0.1191920 * Glin + 0.9503041 * Blin) * 100.0;

  fy := F(Y / Yn);
  fz := F(Z / Zn);

  Result.L := 116.0 * fy - 16.0;
  Result.a := 500.0 * (fx - fy);
  Result.b := 200.0 * (fy - fz);
end;

function RGBToLab(R, G, B: Byte): TLab;
begin
  Result := LinearRGBToLab(SRGBToLinear(R), SRGBToLinear(G), SRGBToLinear(B));
end;

function DegToRad(D: Double): Double;
begin
  Result := D * Pi / 180.0;
end;

  kC = 1.0;
  kH = 1.0;

  Lab2 := RGBToLab(R2, G2, B2);

  C1 := Sqrt(Sqr(Lab1.a) + Sqr(Lab1.b));
  C2 := Sqrt(Sqr(Lab2.a) + Sqr(Lab2.b));

  a2p := Lab2.a * (1.0 + G);
