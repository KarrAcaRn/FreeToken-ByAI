  deltaH := 2.0 * Sqrt(c1p * c2p) * Sin(dh * DEG2RAD * 0.5);

  // --- Step 3: Weighting functions ---
  Lbar := (L1 + L2) / 2.0;

  if (c1p * c2p) = 0.0 then
    hbar := h1p + h2p
  else if Abs(h1p - h2p) <= 180.0 then
    hbar := (h1p + h2p) / 2.0
  else if (h1p + h2p) < 360.0 then
    hbar := (h1p + h2p + 360.0) / 2.0
  else
    hbar := (h1p + h2p - 360.0) / 2.0;

  T := 1.0
     - 0.17 * Cos((hbar - 30.0) * DEG2RAD)
     + 0.24 * Cos(2.0 * hbar * DEG2RAD)
     + 0.32 * Cos((3.0 * hbar + 6.0) * DEG2RAD)
     - 0.20 * Cos((4.0 * hbar - 63.0) * DEG2RAD);

  dtheta := 30.0 * Exp(-Sqr((hbar - 275.0) / 25.0));
  Rc := 2.0 * Sqrt(Power(cbar, 7.0)
       / (Power(cbar, 7.0) + Power(25.0, 7.0)));

  Sl := 1.0 + 0.015 * Sqr(Lbar - 50.0)
             / Sqrt(7.0 + Sqr(Lbar - 50.0));
  Sc := 1.0 + 0.045 * cbar;
  Sh := 1.0 + 0.015 * cbar * T;
  Rt := -Sin(2.0 * dtheta * DEG2RAD) * Rc;

  // --- Final Delta E 2000 (kL = kC = kH = 1) ---
  termL := dLp / Sl;
  termC := dCp / Sc;
  termH := deltaH / Sh;

  Result := Sqrt(Sqr(termL) + Sqr(termC) + Sqr(termH) + Rt * termC * termH);
end;

function DeltaE2000(R1, G1, B1, R2, G2, B2: Byte): Double;
var
  L1, A1, B1v: Double;
  L2, A2, B2v: Double;
begin
  RGBtoLab(R1, G1, B1, L1, A1, B1v);
  RGBtoLab(R2, G2, B2, L2, A2, B2v);
  Result := CIEDE2000(L1, A1, B1v, L2, A2, B2v);
end;

end.
