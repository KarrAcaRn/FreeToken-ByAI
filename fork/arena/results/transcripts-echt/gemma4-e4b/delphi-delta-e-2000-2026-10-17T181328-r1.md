  delta_h_angle := Abs(h1_angle - h2_angle);
  if delta_h_angle > PI then
    delta_h_angle := 2.0 * PI - delta_h_angle;
    
  DeltaH := delta_h_angle;

  // Delta C
  DeltaC := Sqrt(deltaA*deltaA + deltaB*deltaB);
  
  // CIEDE2000 Formel (kL=kC=kH=1)
  Result := Sqrt(
    Math.Pow(deltaL, 2) +
    Math.Pow(DeltaC, 2) +
    Math.Pow(DeltaH, 2)
  );
end;


// Hauptfunktion: Delta E 2000
function DeltaE2000(R1, G1, B1, R2, G2, B2: Byte): Double;
var
  XYZ1, XYZ2: array[1..3] of Double;
  Lab1, Lab2: array[1..3] of Double;
  dE: Double;
begin
  // 1. RGB -> XYZ
  XYZ1 := RGBToXYZ(R1, G1, B1);
  XYZ2 := RGBToXYZ(R2, G2, B2);

  // 2. XYZ -> CIELAB
  Lab1 := XYZToLab(XYZ1[1], XYZ1[2], XYZ1[3]);
  Lab2 := XYZToLab(XYZ2[1], XYZ2[2], XYZ2[3]);

  // 3. CIELAB -> Delta E 2000
  dE := CIEDE2000(
    Lab1[1], Lab1[2], Lab1[3], // L1, a1, b1
    Lab2[1], Lab2[2], Lab2[3]  // L2, a2, b2
  );
  
  Result := dE;
end;

end.
