  fC := f_transform(fY);
  fH := f_transform(fX);
  fL := f_transform(fZ);
  
  // CIELAB Berechnung
  Result.F1 := fL * 116.0 - 16.0; // L*
  Result.F2 := 500.0 * (fH - fC); // a*
  Result.F3 := 200.0 * (fC - fL); // b*
end;


// --- Hauptfunktion: Delta E 2000 ---
function DeltaE2000(R1, G1, B1, R2, G2, B2: Byte): Double;
var
  X1, Y1, Z1, X2, Y2, Z2: Double;
  L1, a1, b1, L2, a2, b2: Double;
  Lab1, Lab2: TTuple<Double, Double, Double>;
  deltaL, deltaa, deltab: Double;
  K1_term, K2_term, K3_term: Double;
begin
  // 1. Konvertierung von sRGB zu XYZ
  X1 := ScaleRGBToXYZ(R1, G1, B1);
  X2 := ScaleRGBToXYZ(R2, G2, B2);

  // 2. Konvertierung von XYZ zu CIELAB
  Lab1 := XYZToCIELAB(X1, Y1, Z1);
  Lab2 := XYZToCIELAB(X2, Y2, Z2);
    
  L1 := Lab1.F1; a1 := Lab1.F2; b1 := Lab1.F3;
  L2 := Lab2.F1; a2 := Lab2.F2; b2 := Lab2.F3;

  // 3. Berechnung von Delta E 2000 (mit kL = kC = kH = 1)
  
  deltaL := L1 - L2;
  deltaa := a1 - a2;
  deltab := b1 - b2;
    
  // DeltaE 2000 Formel (vereinfacht für kL=kC=kH=1)
  
  // K1 Term (L-Komponente)
  K1_num := deltaL * 1.0;
  K1_den := 3.0; // kL^2 + kC^2 + kH^2 = 3
  K1_term := (K1_num / K1_den) * (K1_num / K1_den);

  // K2 Term (a-Komponente)
  K2_num := deltaa * 1.0;
  K2_den := 3.0;
  K2_term := (K2_num / K2_den) * (K2_num / K2_den);

  // K3 Term (b-Komponente)
  K3_num := deltab * 1.0;
  K3_den := 3.0;
  K3_term := (K3_num / K3_den) * (K3_num / K3_den);
    
  // Endgültige Wurzel
  Result := Sqrt(K1_term + K2_term + K3_term);
end;
