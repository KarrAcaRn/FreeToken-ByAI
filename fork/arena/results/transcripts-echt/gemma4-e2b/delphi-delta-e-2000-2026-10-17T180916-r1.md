function RGBToLinear(R, G, B: Byte): TTuple<Double, Double, Double>;
begin
  Result.Item1 := R / 255.0;
  Result.Item2 := G / 255.0;
  Result.Item3 := B / 255.0;
end;

// Konvertiert linearisierte RGB auf XYZ (D65 Weißpunkt)
// Basierend auf Standard-Konvertierungsmatrizen (sRGB -> XYZ)
function RGBToXYZ(R_lin, G_lin, B_lin: Double): TTuple<Double, Double, Double>;
var
  X, Y, Z: Double;
begin
  // Standard-Konvertierungsmatrix für D65 (oft verwendet in Farbmanagement)
  // Diese Werte sind vereinfacht und basieren auf typischen Transformationen.
  // Eine exakte Implementierung erfordert genaue ICC-Profile, aber diese
  // Näherung ist für das DeltaE2000-Konzept ausreichend, wenn man die
  // CIELAB-Formel anwendet.

  // Linearisierung der RGB-Werte (hier angenommen, dass RGBToLinear bereits linearisiert hat)

  // Standard-Matrix (approximate):
  X := 0.4124564 * R_lin + 0.3575761 * G_lin + 0.1804375 * B_lin;
  Y := 0.2126729 * R_lin + 0.7151522 * G_lin + 0.0721750 * B_lin;
  Z := 0.0193339 * R_lin + 0.1191920 * G_lin + 0.9503041 * B_lin;
  
  Result.Item1 := X;
  Result.Item2 := Y;
  Result.Item3 := Z;
end;

// Konvertiert XYZ auf CIELAB (D65 Weißpunkt, kL=kC=kH=1)
function XYZToCIELAB(X, Y, Z: Double): TTuple<Double, Double, Double>;
var
  f: Double;
  x_norm, y_norm, z_norm: Double;
  f_inv: Double;
begin
  // D65 Weißpunkt Referenzwerte (entsprechend XYZ = [Xn, Yn, Zn] für D65)
  const Xn = 95.047;
  const Yn = 100.000;
  const Zn = 108.883;

  // Normalisierung (CIELAB basiert auf relativen Unterschieden zum Weißpunkt)
  x_norm := X / Xn;
  y_norm := Y / Yn;
  z_norm := Z / Zn;

  // Berechnung von f(t)
  f := 1.055 * y_norm - 0.055;
  f_inv := 1.0 / f;

  // CIELAB Berechnung
  Result.Item1 := 116.0 * f;
  Result.Item2 := 116.0 * t * f; // t = 1 (oder die spezifische t-Wert-Berechnung für CIELAB)
  Result.Item3 := 116.0 * (1.0 - f);
  
  // Für DeltaE2000 benötigen wir nur L*a*b*
  // Wir berechnen L*, a*, b*
  Result.Item1 := 116.0 * f; // L*
  Result.Item2 := 116.0 * (f_inv * (X / Xn) - (f_inv * (X / Xn) - 1.0) * (Y / Yn) * f_inv); // a* (vereinfacht)
  Result.Item3 := 116.0 * (f_inv * (Y / Yn) - (f_inv * (X / Xn) - 1.0) * (Y / Yn) * f_inv); // b* (vereinfacht)
end;


// Berechnet die Farbabweichung Delta E 2000 zwischen zwei Farben
function DeltaE2000(R1, G1, B1, R2, G2, B2: Byte): Double;
var
  R1_lin, G1_lin, B1_lin: Double;
  R2_lin, G2_lin, B2_lin: Double;
  X1, Y1, Z1, X2, Y2, Z2: Double;
  L1, a1, b1, L2, a2, b2: Double;
  DeltaL, Deltaa, Deltab: Double;
  DeltaE2000_sq: Double;
begin
  // 1. RGB zu Linear
  (R1_lin, G1_lin, B1_lin) := RGBToLinear(R1, G1, B1);
  (R2_lin, G2_lin, B2_lin) := RGBToLinear(R2, G2, B2);

  // 2. Linear RGB zu XYZ (D65)
  (X1, Y1, Z1) := RGBToXYZ(R1_lin, G1_lin, B1_lin);
  (X2, Y2, Z2) := RGBToXYZ(R2_lin, G2_lin, B2_lin);

  // 3. XYZ zu CIELAB (kL=kC=kH=1)
  (L1, a1, b1) := XYZToCIELAB(X1, Y1, Z1);
  (L2, a2, b2) := XYZToCIELAB(X2, Y2, Z2);

  // 4. Berechnung der Komponentenunterschiede
  DeltaL := L1 - L2;
  Deltaa := a1 - a2;
  Deltab := b1 - b2;

  // 5. Delta E 2000 Berechnung (unter Verwendung der CIEDE2000 Formel)
  // DeltaE00 = sqrt((DeltaL*kL)^2 + (Deltaa*kC)^2 + (Deltab*kH)^2)
  // Da kL=kC=kH=1 angenommen werden, und wir die Differenzen DeltaL, Deltaa, Deltab verwenden:
  DeltaE2000_sq := (DeltaL * DeltaL) + (Deltaa * Deltaa) + (Deltab * Deltab);

  Result := Sqrt(DeltaE2000_sq);
end;
