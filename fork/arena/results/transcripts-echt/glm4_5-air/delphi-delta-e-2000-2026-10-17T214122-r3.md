uses SysUtils, Math;

type
  TXYZ = record
    X, Y, Z: Double;
  end;

  TLAB = record
    L, a, b: Double;
  end;

function RGBToXYZ(R, G, B: Byte): TXYZ;
var
  lr, lg, lb: Double;
begin
  lr := R / 255.0;
  lg := G / 255.0;
  lb := B / 255.0;

  if lr <= 0.04045 then
    lr := lr / 12.92
  else
    lr := Power((lr + 0.055) / 1.055, 2.4);

  if lg <= 0.04045 then
    lg := lg / 12.92
  else
    lg := Power((lg + 0.055) / 1.055, 2.4);

  if lb <= 0.04045 then
    lb := lb / 12.92
  else
    lb := Power((lb + 0.055) / 1.055, 2.4);

  Result.X := (lr * 0.4124564 + lg * 0.3575761 + lb * 0.1804375) * 100;
  Result.Y := (lr * 0.2126729 + lg * 0.7151522 + lb * 0.0721750) * 100;
  Result.Z := (lr * 0.0193339 + lg * 0.1191920 + lb * 0.9503041) * 100;
end;

function XYZToLAB(X, Y, Z: Double): TLAB;
var
  fx, fy, fz: Double;
  Xn, Yn, Zn: Double;
begin
  Xn := 95.047;
  Yn := 100.000;
  Zn := 108.883;

  fx := X / Xn;
  fy := Y / Yn;
  fz := Z / Zn;

  if fx > 0.008856 then
    fx := Power(fx, 1/3)
  else
    fx := (7.787 * fx + 16/116);

  if fy > 0.008856 then
    fy := Power(fy, 1/3)
  else
    fy := (7.787 * fy + 16/116);

    fz := (7.787 * fz + 16/116);

  Result.L := 116 * fy - 16;
  Result.a := 500 * (fx - fy);
  Result.b := 200 * (fy - fz);
end;

  XYZ1 := RGBToXYZ(R1, G1, B1);
  XYZ2 := RGBToXYZ(R2, G2, B2);

  a1 := LAB1.a;
  b1 := LAB1.b;
  L2 := LAB2.L;
  a2 := LAB2.a;
  b2 := LAB2.b;
