```delphi
    dHue := H2 - H1 - 360.0;

  // --- 6. Differenzen ---
  dL := L2v - L1v;
  dC := C2p - C1p;
  dH := 2.0 * Sqrt(C1p * C2p) * Sin(dHue * Pi180 / 2.0);

  // --- 7. Mittelwerte ---
  Lbar := 0.5 * (L1v + L2v);
  Cbar := 0.5 * (C1p + C2p);
  if Abs(H1 - H2) <= 180.0 then
    hbar := 0.5 * (H1 + H2)
  else if (H1 + H2) < 360.0 then
    hbar := 0.5 * (H1 + H2 + 360.0)
  else
    hbar := 0.5 * (H1 + H2 - 360.0);

  // --- 8. T-Ausdruck ---
  T := 1.0
      - 0.17 * Cos((hbar - 30.0) * Pi180)
      + 0.24 * Cos(2.0 * hbar * Pi180)
      + 0.32 * Cos((3.0 * hbar + 6.0) * Pi180)
      - 0.20 * Cos((4.0 * hbar - 63.0) * Pi180);

  // --- 9. Gewichtungsfaktoren (kL = kC = kH = 1) ---
  SL := (Lbar / 100.0)
        + (16.0 * Power(Lbar / 100.0, 3.0))
          / (1.0 + 0.045 * Power(Lbar / 100.0, 3.0));

  SC := 1.0
        + (0.015 * Power(Cbar / 100.0, 2.0))
          / Sqrt(20.0 + Power(Cbar / 100.0, 2.0));

  SH := 1.0
        + 0.048 * Cbar * T
        + (0.015 * Cbar * T) / Sqrt(20.0 + Power(Cbar / 100.0, 2.0));

  TermC := dC / SC;
  TermH := dH / SH;

  // --- 10. Rotationsterm RT ---
  RT := 0.0;
  Theta := 180.0;
  if (hbar >= 120.0) and (hbar < 215.0) then
    Theta := 168.75
  else if (hbar >= 215.0) and (hbar < 325.0) then
    Theta := 275.75
  else if (hbar >= 325.0) and (hbar < 360.0) then
    Theta := 120.0;
  RT := -Sin(2.0 * Theta * Pi180);

  // --- 11. CIEDE2000 ---
  Result := Sqrt(
             Power(dL / SL, 2.0)
           + Power(TermC, 2.0)
           + Power(TermH, 2.0)
           + RT * TermC * TermH);
end;

end.
```

Der Kernfehler war: In Free Pascal (Delphi-Modus) ist das Schatten lokaler Variablen durch Parameter **nicht zulässig**. Die lokalen Variablen `b1`/`b2` kollidierten deshalb mit den Parametern `B1`/`B2` der Funktion `DeltaE2000` → „Duplicate identifier". Dadurch wurde `b1`/`b2` nicht als Double erkannt, sondern als der Byte-Parameter `B1`/`B2` → „Incompatible type … Got Double, expected Byte".

Fix: Umbenennung der lokalen Variablen in `L1v, a1v, b1v, L2v, a2v, b2v` (keine Kollision mehr mit `B1`/`B2`) sowie der `var`-Parameter in `RGBToLab` zu `Lr, ar, br`.
