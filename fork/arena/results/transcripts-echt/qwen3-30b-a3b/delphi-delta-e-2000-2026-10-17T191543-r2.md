procedure XYZToCIELAB(X, Y, Z: Double; var L, a, b: Double);
function DeltaE2000(R1, G1, B1, R2, G2, B2: Byte): Double;

implementation

const
  Xn = 0.95047;
  Yn = 1.0;
  Zn = 1.08883;
  k = 0.008856;
  k1 = 7.787;
  k2 = 16/116;
  k3 = 116;
  k4 = 500;
  k5 = 200;

procedure RGBToXYZ(R, G, B: Byte; var X, Y, Z: Double);
var
  R_linear, G_linear, B_linear: Double;
begin
  R_linear := R / 255;
  if R_linear > 0.04045 then
    R_linear := Power((R_linear + 0.055) / 1.055, 2.4)
  else
    R_linear := R_linear / 12.92;

  G_linear := G / 255;
  if G_linear > 0.04045 then
    G_linear := Power((G_linear + 0.055) / 1.055, 2.4)
  else
    G_linear := G_linear / 12.92;

  B_linear := B / 255;
  if B_linear > 0.04045 then
    B_linear := Power((B_linear + 0.055) / 1.055, 2.4)
  else
    B_linear := B_linear / 12.92;

  X := 0.4124564 * R_linear + 0.3575761 * G_linear + 0.1804375 * B_linear;
  Y := 0.2126729 * R_linear + 0.7151522 * G_linear + 0.0721729 * B_linear;
  Z := 0.0193339 * R_linear + 0.1191930 * G_linear + 0.9504479 * B_linear;
end;

procedure XYZToCIELAB(X, Y, Z: Double; var L, a, b: Double);
var
  fx, fy, fz: Double;
begin
  function F(x: Double): Double;
  begin
    if x > k then
      Result := Power(x, 1/3)
    else
      Result := k1 * x + k2;
  end;

  fx := F(X / Xn);
  fy := F(Y / Yn);
  fz := F(Z / Zn);

  L := k3 * fy - 16;
  a := k4 * (fx - fy);
  b := k5 * (fy - fz);
end;

function DeltaE2000(R1, G1, B1, R2, G2, B2: Byte): Double;
var
  X1, Y1, Z1, X2, Y2, Z2: Double;
  L1, a1, b1, L2, a2, b2: Double;
  C1, C2, H1, H2: Double;
  Lbar, Cbar, Hbar: Double;
  deltaL, deltaC, deltaH: Double;
  deltaE: Double;
begin
  // Convert RGB to XYZ
  RGBToXYZ(R1, G1, B1, X1, Y1, Z1);
  RGBToXYZ(R2, G2, B2, X2, Y2, Z2);

  // Convert XYZ to CIELAB
  XYZToCIELAB(X1, Y1, Z1, L1, a1, b1);
  XYZToCIELAB(X2, Y2, Z2, L2, a2, b2);

  // Calculate chroma and hue
  C1 := Sqrt(a1*a1 + b1*b1);
  C2 := Sqrt(a2*a2 + b2*b2);
  H1 := ArcTan2(b1, a1) * 180 / Pi;
  if H1 < 0 then H1 := H1 + 360;
  H2 := ArcTan2(b2, a2) * 180 / Pi;
  if H2 < 0 then H2 := H2 + 360;

  // Calculate average values
  Lbar := (L1 + L2) / 2;
  Cbar := (C1 + C2) / 2;
  Hbar := (H1 + H2) / 2;
  if Abs(H1 - H2) > 180 then
    Hbar := (H1 + H2) / 2;
  if Hbar > 360 then Hbar := Hbar - 360;
  if Hbar < 0 then Hbar := Hbar + 360;

  // Compute deltaL, deltaC, deltaH
  deltaL := L1 - L2;
  deltaC := C1 - C2;
  deltaH := H1 - H2;

  // Normalize deltaH to 0-360
  if deltaH > 180 then deltaH := deltaH - 360;
  if deltaH < -180 then deltaH := deltaH + 360;

  // Calculate Delta E
  deltaE := Sqrt(deltaL*deltaL + deltaC*deltaC + deltaH*deltaH);
  Result := deltaE;
end;

end.
