  Result := R * 180.0 / Pi;
  while Result < 0.0 do Result := Result + 360.0;
  while Result >= 360.0 do Result := Result - 360.0;
end;

{ ---------- CIEDE2000 (kL = kC = kH = 1) ---------- }

function CIEDE2000(L1, a1, b1, L2, a2, b2: Double): Double;
const
  K7 = 6103515625.0;
var
  C1, C2, Cbar, G, Gfactor: Double;
  a1p, a2p: Double;
  C1p, C2p: Double;
  h1p, h2p: Double;
  dLp, dCp, dhue, dHueW: Double;
  Lbp, Cbp, hbp: Double;
  T, dTheta, RC: Double;
  SL, SC, SH: Double;
  RT: Double;
begin
  { 1 – Chroma }
  C1 := Sqrt(a1 * a1 + b1 * b1);
  C2 := Sqrt(a2 * a2 + b2 * b2);
  Cbar := (C1 + C2) * 0.5;

  { 2 – G-Faktor }
  Gfactor := Power(Cbar, 7);
  G := 0.5 * (1.0 - Sqrt(Gfactor / (Gfactor + K7)));

  { 3 – verzerrtes a' }
  a1p := a1 * (1.0 + G);
  a2p := a2 * (1.0 + G);

  { 4 – neue Chroma }
  C1p := Sqrt(a1p * a1p + b1 * b1);
  C2p := Sqrt(a2p * a2p + b2 * b2);

  { 5 – Farbtonwinkel }
  if (a1p = 0.0) and (b1 = 0.0) then
    h1p := 0.0
  else
    h1p := RadToDeg(ArcTan2(b1, a1p));

  if (a2p = 0.0) and (b2 = 0.0) then
    h2p := 0.0
  else
    h2p := RadToDeg(ArcTan2(b2, a2p));

  { 6 – Differenzen }
  dLp := L2 - L1;
  dCp := C2p - C1p;

  if (C1p * C2p) = 0.0 then
    dhue := 0.0
  else
  begin
    dhue := h2p - h1p;
    if dhue > 180.0 then dhue := dhue - 360.0
    else if dhue < -180.0 then dhue := dhue + 360.0;
  end;

  dHueW := 2.0 * Sqrt(C1p * C2p) * Sin(DegToRad(dhue * 0.5));

  { 7 – Mittelwerte }
  Lbp := (L1 + L2) * 0.5;
  Cbp := (C1p + C2p) * 0.5;

  if (C1p * C2p) = 0.0 then
    hbp := h1p + h2p
  else
  begin
    hbp := (h1p + h2p) * 0.5;
    if Abs(h1p - h2p) > 180.0 then
    begin
      if (h1p + h2p) < 360.0 then
        hbp := (h1p + h2p + 360.0) * 0.5
      else
        hbp := (h1p + h2p - 360.0) * 0.5;
    end;
  end;

  { 8 – Rotations-/Gewichtsterme }
  T := 1.0
     - 0.17 * Cos(DegToRad(hbp - 30.0))
     + 0.24 * Cos(DegToRad(2.0 * hbp))
     + 0.32 * Cos(DegToRad(3.0 * hbp + 6.0))
     - 0.20 * Cos(DegToRad(4.0 * hbp - 63.0));

  dTheta := 30.0 * Exp(-Power((hbp - 275.0) / 25.0, 2.0));

  RC := 2.0 * Sqrt(Power(Cbp, 7) / (Power(Cbp, 7) + K7));

  SL := 1.0 + (0.015 * Sqr(Lbp - 50.0)) / Sqrt(20.0 + Sqr(Lbp - 50.0));
  SC := 1.0 + 0.045 * Cbp;
  SH := 1.0 + 0.015 * Cbp * T;

  RT := -Sin(DegToRad(2.0 * dTheta)) * RC;

  { 9 – Delta E 2000 }
  Result := Sqrt(
    Sqr(dLp / SL) +
    Sqr(dCp / SC) +
    Sqr(dHueW / SH) +
    RT * (dCp / SC) * (dHueW / SH)
  );
end;

{ ---------- Öffentliche Funktion ---------- }

function DeltaE2000(R1, G1, B1, R2, G2, B2: Byte): Double;
var
  LabL1, LabA1, LabB1, LabL2, LabA2, LabB2: Double;
begin
  RGBToLab(R1, G1, B1, LabL1, LabA1, LabB1);
  RGBToLab(R2, G2, B2, LabL2, LabA2, LabB2);
  Result := CIEDE2000(LabL1, LabA1, LabB1, LabL2, LabA2, LabB2);
end;

end.
