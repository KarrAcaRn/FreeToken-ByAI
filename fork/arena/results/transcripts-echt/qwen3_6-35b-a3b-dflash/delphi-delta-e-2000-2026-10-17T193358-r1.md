const
  // D65 White Point
  Xn = 0.95047;
  Yn = 1.00000;
  Zn = 1.08883;
  
  // sRGB to XYZ Matrix (Row 1, 2, 3)
  // R       G       B
  M1 = 0.4124564; M2 = 0.3575761; M3 = 0.1804375;

  M7 = 0.0193339; M8 = 0.1191920; M9 = 0.9503041;
  
  // PI constant
  PI = 3.14159265358979;
  
var
  // Helper to linearize sRGB value (0-1) to linear light
  function LinearizeRGB(Value: Double): Double;
  begin
    // sRGB standard:
    // If V <= 0.04045, V_lin = V / 12.92
    // Else V_lin = ((V + 0.055) / 1.055)^2.4
    if Value <= 0.04045 then
      Result := Value / 12.92
    else
      Result := Power((Value + 0.055) / 1.055, 2.4);
  end;

  // Helper for CIELAB f(t) function
  function F_Lab(t: Double): Double;
  begin
    // (6/29)^3 = 0.00885645167...
    if t > 0.00885645167 then
      Result := Power(t, 1.0 / 3.0)
    else
      Result := 7.787037 * t + 0.137931;
  end;

  X1, Y1, Z1, X2, Y2, Z2: Double;

  var g1 := G1 / 255.0;
  var b1_rgb := B1 / 255.0; // renamed to avoid conflict with b1 (Lab)

  a2 := 500.0 * (fx2 - fy2);
  b2 := 200.0 * (fy2 - fz2);

  // L_bar
  L_bar := (L1 + L2) / 2.0;
  
  // Chroma
  C1 := Sqrt(a1 * a1 + b1 * b1);
  C2 := Sqrt(a2 * a2 + b2 * b2);
  C_bar := (C1 + C2) / 2.0;
  
  // G factor
  // G = 0.5 * (1 - sqrt(C_bar^7 / (C_bar^7 + 25^7)))
  // 25^7 = 6103515625
  G := 0.5 * (1.0 - Sqrt(Power(C_bar, 7) / (Power(C_bar, 7) + 6103515625.0)));
  
  // Adjusted a*
  a1p := a1 * (1.0 + G);
  a2p := a2 * (1.0 + G);
  
  // Adjusted Chroma
  Cp1 := Sqrt(a1p * a1p + b1 * b1);
  Cp2 := Sqrt(a2p * a2p + b2 * b2);
  
  // Hue angles in degrees
  h1p := ArcTan2(b1, a1p) * 180.0 / PI;
  h2p := ArcTan2(b2, a2p) * 180.0 / PI;

  if h1p < 0.0 then h1p := h1p + 360.0;
  if h2p < 0.0 then h2p := h2p + 360.0;

  // T = 1 - 0.17 * Cos(h1p - 30) + 0.13 * Cos(2*h1p) + 0.1 * Cos(3*h1p) - 0.06 * Cos(4*h1p)
  // Angles must be converted to radians for Cos function
  T_val := 1.0
           - 0.17 * Cos((h1p - 30.0) * PI / 180.0)
           + 0.13 * Cos(2.0 * h1p * PI / 180.0)
           + 0.10 * Cos(3.0 * h1p * PI / 180.0)
           - 0.06 * Cos(4.0 * h1p * PI / 180.0);

  // We need the shortest arc difference for calculation
  if Abs(h1p - h2p) <= 180.0 then
    h_bar := (h1p + h2p) / 2.0
  else
  begin
    if h1p + h2p < 360.0 then
      h_bar := (h1p + h2p) / 2.0 + 180.0
    else
      h_bar := (h1p + h2p) / 2.0 - 180.0;
  end;
  // Normalize h_bar
  if h_bar >= 360.0 then h_bar := h_bar - 360.0;
  if h_bar < 0.0 then h_bar := h_bar + 360.0;
  
  // Delta L' and Delta C'
  delta_L_prime := L2 - L1;
  delta_C_prime := Cp2 - Cp1;
  
  // Delta H'
  // Delta H' = 2 * sqrt(Cp1 * Cp2) * sin((h2p - h1p)/2)
  // The angle difference must be the shortest arc angle
  diff_h := h2p - h1p;
  if diff_h > 180.0 then diff_h := diff_h - 360.0;
  if diff_h < -180.0 then diff_h := diff_h + 360.0;
  
  delta_H_prime := 2.0 * Sqrt(Cp1 * Cp2) * Sin(diff_h * PI / 180.0 / 2.0);
  
  // Weighting functions
  // S_L
  S_L := 1.0 + (0.015 * Power(L_bar - 50.0, 2.0)) / Sqrt(20.0 + Power(L_bar - 50.0, 2.0));
  
  // S_C
  S_C := 1.0 + 0.045 * C_bar;
  
  // S_H
  S_H := 1.0 + 0.015 * C_bar * T_val;
  
  // Final Delta E 2000
  Result := Sqrt(
    Power(delta_L_prime / S_L, 2.0) +
    Power(delta_C_prime / S_C, 2.0) +
    Power(delta_H_prime / S_H, 2.0)
  );
end;
