  c1_prime := Sqrt(Sqr(a1_prime) + Sqr(lab1.b));
  c2_prime := Sqrt(Sqr(a2_prime) + Sqr(lab2.b));

  // Schritt 6: Hue Winkel (in Grad)
  h1_prime := IfThen((a1_prime = 0) and (lab1.b = 0), 0, 
                   RadToDeg(ArcTan2(lab1.b, a1_prime)));
  h2_prime := IfThen((a2_prime = 0) and (lab2.b = 0), 0, 
                   RadToDeg(ArcTan2(lab2.b, a2_prime)));

  // Schritt 7: Hue Differenz
  delta_h_prime := h2_prime - h1_prime;
  if Abs(delta_h_prime) <= 180 then
    delta_h_prime := delta_h_prime
  else if delta_h_prime > 180 then
    delta_h_prime := delta_h_prime - 360
  else
    delta_h_prime := delta_h_prime + 360;

  // Schritt 8: Delta L, C, H
  delta_l := lab2.L - lab1.L;
  delta_c := c2_prime - c1_prime;
  delta_h := 2 * Sqrt(c1_prime * c2_prime) * Sin(DegToRad(delta_h_prime / 2));

  // Schritt 9: Mittelwerte
  l_mean := (lab1.L + lab2.L) / 2;
  c_mean_prime := (c1_prime + c2_prime) / 2;

  // Schritt 10: Mittelwert Hue
  if Abs(h1_prime - h2_prime) <= 180 then
    h_mean := (h1_prime + h2_prime) / 2
  else
  begin
    if (h1_prime + h2_prime) < 0 then
      h_mean := (h1_prime + h2_prime + 360) / 2
    else
      h_mean := (h1_prime + h2_prime - 360) / 2;
  end;

  // Schritt 11: T-Faktor
  t := 1 - 0.17 * Cos(DegToRad(h_mean - 30)) + 
       0.24 * Cos(DegToRad(2 * h_mean)) + 
       0.32 * Cos(DegToRad(3 * h_mean + 6)) - 
       0.20 * Cos(DegToRad(4 * h_mean - 63));

  // Schritt 12: Delta Theta
  delta_theta := 30 * Exp(-Sqr((h_mean - 275) / 25));

  // Schritt 13: R_C
  r_c := 2 * Sqrt(Power(c_mean_prime, 7) / (Power(c_mean_prime, 7) + Power(25, 7)));

  // Schritt 14: S_L, S_C, S_H
  s_l := 1 + (0.015 * Sqr(l_mean - 50)) / Sqrt(20 + Sqr(l_mean - 50));
  s_c := 1 + 0.045 * c_mean_prime;
  s_h := 1 + 0.015 * c_mean_prime * r_c;

  // Schritt 15: R_T
  rt := -Sin(DegToRad(2 * delta_theta)) * r_c;

  // Schritt 16: Delta E 2000
  Result := Sqrt(
    Sqr(delta_l / s_l) + 
    Sqr(delta_c / s_c) + 
    Sqr(delta_h / s_h) + 
    rt * (delta_c / s_c) * (delta_h / s_h)
  );
end;
