procedure XYZToLAB(X, Y, Z: Double; var L, a, b: Double);

  Xn = 95.047;
  Yn = 100.000;
  Zn = 108.883;
  Epsilon = 0.008856;
  Kappa   = 7.787037037;
  Four29  = 0.137931034;

  fx, fy, fz: Double;

  if Y / Yn > Epsilon then fy := CubicRoot(Y / Yn) else fy := Kappa * (Y / Yn) + Four29;
  if Z / Zn > Epsilon then fz := CubicRoot(Z / Zn) else fz := Kappa * (Z / Zn) + Four29;
  L := 116.0 * fy - 16.0;
  a := 500.0 * (fx - fy);
  b := 200.0 * (fy - fz);
end;

  X1, Y1, Z1, X2, Y2, Z2: Double;
  L1, a1, b1, L2, a2, b2: Double;
  c1, c2: Double;
  cbar, G: Double;
  a1p, a2p, c1p, c2p: Double;
  h1, h2: Double;
  dLp, dcp, dhp, dhpRad: Double;
  Lbar, cbarp, hbar: Double;
  T, dtheta, Rc, Rt, Sl, Sc, Sh: Double;
begin
  sRGBToXYZ(R1, G1, B1, X1, Y1, Z1);
  sRGBToXYZ(R2, G2, B2, X2, Y2, Z2);

  XYZToLAB(X1, Y1, Z1, L1, a1, b1);
  XYZToLAB(X2, Y2, Z2, L2, a2, b2);

  // Chroma
  c1 := Sqrt(a1 * a1 + b1 * b1);
  c2 := Sqrt(a2 * a2 + b2 * b2);

  // G factor
  cbar := (c1 + c2) / 2.0;

  // Transformed a*

  a2p := (1.0 + G) * a2;

  c1p := Sqrt(a1p * a1p + b1 * b1);
  c2p := Sqrt(a2p * a2p + b2 * b2);

  // Hue angles

  begin
    h1 := ArcTan2(b1, a1p) * 180.0 / Pi;
    if h1 < 0.0 then h1 := h1 + 360.0;
  end
  else
    h1 := 0.0;

  if c2p > 0 then
  begin
    h2 := ArcTan2(b2, a2p) * 180.0 / Pi;
    if h2 < 0.0 then h2 := h2 + 360.0;
  end
  else
    h2 := 0.0;

  // Differences
  dLp := L2 - L1;
  dcp := c2p - c1p;

  if (h1 = h2) then
    dhp := 0.0
  else if Abs(h2 - h1) <= 180.0 then
    dhp := h2 - h1
  else if h2 > h1 then
    dhp := h2 - h1 - 360.0
  else
    dhp := h2 - h1 + 360.0;

  dhpRad := Sqrt(c1p * c2p) * Sin(dhp * Pi / 180.0);

  // Averages
  Lbar := (L1 + L2) / 2.0;
  cbarp := (c1p + c2p) / 2.0;

  if cbarp > 0 then
  begin
    if Abs(h2 - h1) <= 180.0 then
      hbar := (h1 + h2) / 2.0
    else if (h1 + h2) < 360.0 then
      hbar := (h1 + h2 + 360.0) / 2.0
    else
      hbar := (h1 + h2 - 360.0) / 2.0;
  end
  else
    hbar := 0.0;

  // Weights
  T := 1.0
       - 0.17 * Cos(hbar - 30.0)
       + 0.24 * Cos(2.0 * hbar)
       + 0.32 * Cos(3.0 * hbar + 6.0)
       - 0.20 * Cos(4.0 * hbar - 63.0);

  dtheta := 30.0 * Exp(-Power((hbar - 275.0) / 25.0, 2.0));

  Rc := 2.0 * Sqrt(Power(cbarp, 7) / (Power(cbarp, 7) + Power(25.0, 7)));
  Rt := -Sin(2.0 * dtheta * Pi / 180.0) * Rc;

  Sl := 1.0 + (0.015 * Power(Lbar - 50.0, 2.0)) / Sqrt(20.0 + Power(Lbar - 50.0, 2.0));
  Sc := 1.0 + 0.045 * cbarp;
  Sh := 1.0 + 0.015 * cbarp * T;

  Result := Sqrt(
      Power(dLp / Sl, 2.0)
    + Power(dcp / Sc, 2.0)
    + Power(dhpRad / Sh, 2.0)
    - Rt * (dcp / Sc) * (dhpRad / Sh)
  );
end;

    Result := Power((v + 0.055) / 1.055, 2.4);
end;

// sRGB (8-bit) -> lineare RGB -> XYZ (D65)
procedure sRGBToXYZ(R, G, B: Byte; var X, Y, Z: Double);
var
  rl, gl, bl: Double;
begin
  rl := sRGBToLinear(R);
  gl := sRGBToLinear(G);
  bl := sRGBToLinear(B);
  X := rl * 0.4124564 + gl * 0.3575761 + bl * 0.1804375;
  Y := rl * 0.2126729 + gl * 0.7151522 + bl * 0.0721750;
  Z := rl * 0.0193339 + gl * 0.1191920 + bl * 0.9503041;
end;

// XYZ (D65) -> CIELAB
procedure XYZToLAB(X, Y, Z: Double; var L, a, b: Double);
const
  Xn   = 95.047;
  Yn   = 100.000;
  Zn   = 108.883;
  Eps  = 0.008856;         // (6/29)^3
  Kap  = 7.787037037;      // (1/3)*(29/6)^2
  VIER = 0.137931034;      // 4/29
var
  fx, fy, fz: Double;
begin
  // Kubische Wurzel via Power (CubicRoot ggf. nicht in allen Math-Units verfügbar)
  if X / Xn > Eps then fx := Power(X / Xn, 1.0 / 3.0) else fx := Kap * (X / Xn) + VIER;
  if Y / Yn > Eps then fy := Power(Y / Yn, 1.0 / 3.0) else fy := Kap * (Y / Yn) + VIER;
  if Z / Zn > Eps then fz := Power(Z / Zn, 1.0 / 3.0) else fz := Kap * (Z / Zn) + VIER;
  L := 116.0 * fy - 16.0;
  a := 500.0 * (fx - fy);
  b := 200.0 * (fy - fz);
end;

function DeltaE2000(R1, G1, B1, R2, G2, B2: Byte): Double;
var
  X1, Y1, Z1, X2, Y2, Z2: Double;
  L1, a1, b1v, L2, a2, b2v: Double;
  c1, c2, cbar, cbarp, G: Double;
  a1p, a2p, c1p, c2p: Double;
  h1, h2, dLp, dcp, dhp, dhpRad: Double;
  Lbar, hbar, T, dtheta, Rc, Rt, Sl, Sc, Sh: Double;
begin
  // 1) 8-bit sRGB -> lineare RGB -> XYZ (D65)
  sRGBToXYZ(R1, G1, B1, X1, Y1, Z1);
  sRGBToXYZ(R2, G2, B2, X2, Y2, Z2);

  // 2) XYZ -> CIELAB
  XYZToLAB(X1, Y1, Z1, L1, a1, b1v);
  XYZToLAB(X2, Y2, Z2, L2, a2, b2v);

  // 3) CIEDE2000 (kL = kC = kH = 1)
  c1 := Sqrt(a1 * a1 + b1v * b1v);
  c2 := Sqrt(a2 * a2 + b2v * b2v);

  // G-Faktor
  cbar := (c1 + c2) / 2.0;
  G := 0.5 * (1.0 - Sqrt(Power(cbar, 7) / (Power(cbar, 7) + Power(25.0, 7))));

  // Transformiertes a*
  a1p := (1.0 + G) * a1;
  a2p := (1.0 + G) * a2;

  c1p := Sqrt(a1p * a1p + b1v * b1v);
  c2p := Sqrt(a2p * a2p + b2v * b2v);

  // Helligkeitsdifferenz und Chroma-Differenz
  dLp := L2 - L1;
  dcp := c2p - c1p;

  // Winkel h* (Grad, 0..360)
  if c1p > 0.0 then
  begin
    h1 := ArcTan2(b1v, a1p) * 180.0 / Pi;
    if h1 < 0.0 then h1 := h1 + 360.0;
  end
  else
    h1 := 0.0;

  if c2p > 0.0 then
  begin
    h2 := ArcTan2(b2v, a2p) * 180.0 / Pi;
    if h2 < 0.0 then h2 := h2 + 360.0;
  end
  else
    h2 := 0.0;

  // Differenz der Winkel (als Sinus-Term)
  if (c1p = 0.0) or (c2p = 0.0) then
    dhp := 0.0
  else if Abs(h2 - h1) <= 180.0 then
    dhp := h2 - h1
  else if h2 > h1 then
    dhp := h2 - h1 - 360.0
  else
    dhp := h2 - h1 + 360.0;

  dhpRad := 2.0 * Sqrt(c1p * c2p) * Sin(dhp * Pi / 360.0);

  // Mittelwerte
  Lbar := (L1 + L2) / 2.0;
  cbarp := (c1p + c2p) / 2.0;

  if cbarp > 0.0 then
  begin
    if Abs(h2 - h1) <= 180.0 then
      hbar := (h1 + h2) / 2.0
    else if (h1 + h2) < 360.0 then
      hbar := (h1 + h2 + 360.0) / 2.0
    else
      hbar := (h1 + h2 - 360.0) / 2.0;
  end
  else
    hbar := 0.0;

  // Gewichte
  T := 1.0
       - 0.17 * Cos(hbar - 30.0)
       + 0.24 * Cos(2.0 * hbar)
       + 0.32 * Cos(3.0 * hbar + 6.0)
       - 0.20 * Cos(4.0 * hbar - 63.0);

  dtheta := 30.0 * Exp(-(((hbar - 275.0) / 25.0) * ((hbar - 275.0) / 25.0)));

  Rc := 2.0 * Sqrt(Power(cbarp, 7) / (Power(cbarp, 7) + Power(25.0, 7)));
  Rt := -Sin(2.0 * dtheta * Pi / 180.0) * Rc;

  Sl := 1.0 + (0.015 * ((Lbar - 50.0) * (Lbar - 50.0))) / Sqrt(20.0 + ((Lbar - 50.0) * (Lbar - 50.0)));
  Sc := 1.0 + 0.045 * cbarp;
  Sh := 1.0 + 0.015 * cbarp * T;

  Result := Sqrt(
      (dLp / Sl) * (dLp / Sl)
    + (dcp / Sc) * (dcp / Sc)
    + (dhpRad / Sh) * (dhpRad / Sh)
    - Rt * (dcp / Sc) * (dhpRad / Sh)
  );
end;
