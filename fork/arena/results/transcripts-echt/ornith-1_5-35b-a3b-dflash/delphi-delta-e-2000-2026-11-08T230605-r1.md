var

begin
  C := Value / 255.0;
  if C <= 0.04045 then
    Result := C / 12.92
  else
    Result := Power((C + 0.055) / 1.055, 2.4);
end;

/// <summary>Konvertiert 8-Bit-sRGB-Werte in den CIE 1931 XYZ-Raum (D65).</summary>
procedure RGBtoXYZ(R, G, B: Byte; var X, Y, Z: Double);
var

  g := SRGBToLinear(G);
  b := SRGBToLinear(B);
  X := r * 0.4124564 + g * 0.3575761 + b * 0.1804375;
  Y := r * 0.2126729 + g * 0.7151522 + b * 0.0721750;

end;

end;
