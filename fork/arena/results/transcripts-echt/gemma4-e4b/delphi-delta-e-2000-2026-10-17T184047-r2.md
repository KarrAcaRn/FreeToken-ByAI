function linearRGBtoXYZ(R_Lin, G_Lin, B_Lin: Double): array[1..3] of Double;
const
  Xn = 0.95047;
  Yn = 1.00000;
  Zn = 1.08883;

  // Matrixkoeffizienten für sRGB -> XYZ (D65)
  M_R := 0.4124564;
  M_G := 0.3575761;
  M_B := 0.1804375;

  M_R2 := 0.2126729;
  M_G2 := 0.7151522;
  M_B2 := 0.0721750;

  M_R3 := 0.0193339;
  M_G3 := 0.0119161;
  M_B3 := 0.0115817;
begin
  // Anwendung der sRGB zu XYZ Transformation
  Result[1] := M_R * R_Lin + M_G * G_Lin + M_B * B_Lin;
  Result[2] := M_R2 * R_Lin + M_G2 * G_Lin + M_B2 * B_Lin;
  Result[3] := M_R3 * R_Lin + M_G3 * G_Lin + M_B3 * B_Lin;
end;

// Konvertiert XYZ zu CIELAB (Double)
function XYZtoCIELAB(X, Y, Z: Double): array[1..3] of Double;
const
  Xn = 0.95047;
  Yn = 1.00000;
  Zn = 1.08883;
  epsilon = 0.008856; // 216/24389

  // f(t) Funktion
  function f(t: Double): Double;
  begin
    if t > epsilon then
      Result := Power(t, 1.0/3.0)
    else
      Result := (1.0/3.0) * (16.0 * epsilon)
  end;
begin
  var L_star, a_star, b_star: Double;

  // L*
  L_star := 116.0 * f(Y / Yn) - 16.0;

  // a*
  a_star := 500.0 * (f(X / Xn) - f(Y / Yn));

  // b*
  b_star := 200.0 * (f(Y / Yn) - f(Z / Zn));

  Result[1] := L_star;
  Result[2] := a_star;
  Result[3] := b_star;
end;

function DeltaE2000(R1, G1, B1, R2, G2, B2: Byte): Double;
var
  // Farbkette 1
  R1_Lin, G1_Lin, B1_Lin: Double;
  XYZ1: array[1..3] of Double;
  L1, a1, b1: Double;

  // Farbkette 2
  R2_Lin, G2_Lin, B2_Lin: Double;
  XYZ2: array[1..3] of Double;
  L2, a2, b2: Double;

  // CIEDE2000 Berechnungsvariablen
  LBar, ABar, BBar: Double;
  C1, C2: Double;
  h1, h2: Double;
  DeltaL: Double;
  DeltaC: Double;
  DeltaH: Double;
  DeltaE_2000: Double;
  K_L, K_C, K_H: Double;
begin
  // 1. Konvertierung RGB -> Linear
  R1_Lin := sRGBtoLinear(R1);
  G1_Lin := sRGBtoLinear(G1);
  B1_Lin := sRGBtoLinear(B1);

  R2_Lin := sRGBtoLinear(R2);
  G2_Lin := sRGBtoLinear(G2);
  B2_Lin := sRGBtoLinear(B2);

  // 2. Konvertierung Linear RGB -> XYZ (D65)
  XYZ1 := linearRGBtoXYZ(R1_Lin, G1_Lin, B1_Lin);
  XYZ2 := linearRGBtoXYZ(R2_Lin, G2_Lin, B2_Lin);

  // 3. Konvertierung XYZ -> CIELAB
  // Die Funktion XYZtoCIELAB gibt ein array[1..3] zurück. Wir nutzen die Indexierung [1], [2], [3]
  array[1..3] of Double := XYZtoCIELAB(XYZ1[1], XYZ1[2], XYZ1[3]);
  L1 := array[1..3] of Double[1];
  a1 := array[1..3] of Double[2];
  b1 := array[1..3] of Double[3];

  // Korrekte Zuweisung der einzelnen Werte aus dem Array-Rückgabewert:
  L1 := XYZtoCIELAB(XYZ1[1], XYZ1[2], XYZ1[3])[1];
  a1 := XYZtoCIELAB(XYZ1[1], XYZ1[2], XYZ1[3])[2];
  b1 := XYZtoCIELAB(XYZ1[1], XYZ1[2], XYZ1[3])[3];


  L2 := XYZtoCIELAB(XYZ2[1], XYZ2[2], XYZ2[3])[1];
  a2 := XYZtoCIELAB(XYZ2[1], XYZ2[2], XYZ2[3])[2];
  b2 := XYZtoCIELAB(XYZ2[1], XYZ2[2], XYZ2[3])[3];


  // 4. Berechnung von L_bar, a_bar, b_bar (Mittelwerte)
  LBar := (L1 + L2) / 2.0;
  ABar := (a1 + a2) / 2.0;
  BBar := (b1 + b2) / 2.0;

  // 5. Berechnung von C* und h* für beide Farben
  C1 := Sqrt(a1*a1 + b1*b1);
  C2 := Sqrt(a2*a2 + b2*b2);

  // h = atan(b/a) in Grad
  if C1 > 0.001 then
    h1 := math.Atan(b1 / a1) * (180.0 / math.Pi)
  else
    h1 := 0.0;

  if C2 > 0.001 then
    h2 := math.Atan(b2 / a2) * (180.0 / math.Pi)
  else
    h2 := 0.0;

  // Normalisierung von DeltaH
  DeltaH := math.Abs(h2 - h1);
  if (math.Abs(h2 - h1) > 180.0) then
    DeltaH := 360.0 - math.Abs(h2 - h1);

  // 6. Berechnung der Delta-Komponenten
  DeltaL := L2 - L1;
  DeltaC := C2 - C1;

  // 7. CIEDE2000 (Vereinfachte Form für kL=kC=kH=1)
  DeltaE_2000 := Sqrt(
    Power(DeltaL, 2) +
    Power(DeltaC, 2) +
    Power(DeltaH, 2)
  );

  Result := DeltaE_2000;
end;
