  else
  begin
    da := (da * (L1 + L2 + 12.0)) / (L1 - L2);
    db := (db * (L1 + L2 + 12.0)) / (L1 - L2);
  end;

  term1 := (dL + da * L1 + db * L2) / (L1 + L2 + 12.0);
  term2 := term1 - (da / a1 + db / b1);

  e := 500.0 * Sqrt( (term1 * term1) + (term2 * term2) );

  Result := e;
end;
